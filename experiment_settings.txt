These are the default parameters used for most SIRS experiments
n = 200  # number of nodes
density_contact = 0.05  # probability of edge existence
density_social = 2 * density_contact
beta = 0.15  # Infection rate
gamma = 0.07  # Recovery rate
mu = 0.05  # Immunity loss rate
init = 0.05 # Initial infected portion
T = 100  # Total simulation time
split_point = 30 # Time when quarantining begins
activation_probability = 0.05  # Probability an individual is informed by its neighbor (social network; independent cascade)
num_trials = 25  # Number of times experiments are run, for averaging
